 run(['python3',str(r/'snapshot.py'),label],stdout=subprocess.DEVNULL)
 return json.loads((r/('snapshot-'+label+'.json')).read_text())

 run(['python3',str(r/'wait_health.py'),label,release]);run(['python3',str(r/'capture_phase.py'),label,release]);return json.loads((r/('snapshot-'+label+'.json')).read_text())
