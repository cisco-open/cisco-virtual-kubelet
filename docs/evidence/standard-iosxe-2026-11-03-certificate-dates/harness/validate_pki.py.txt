"""Read-only date/chain/name controls using actual public leaf certificates."""
import pathlib,json,subprocess,datetime
R=pathlib.Path('/tmp/cvk-stddate-20261103');pki=json.loads((R/'pki-manifest.json').read_text());now=datetime.datetime.now(datetime.timezone.utc);results=[]
for name,mode,epoch,expected in [('valid-now','valid',int(now.timestamp()),True),('expired-now','expired',int(now.timestamp()),False),('future-now','future',int(now.timestamp()),False)]+[(mode+'-valid-within-own-window',mode,int((datetime.datetime.fromisoformat(pki['leaves'][mode]['validFromUTC'])+(datetime.datetime.fromisoformat(pki['leaves'][mode]['validUntilUTC'])-datetime.datetime.fromisoformat(pki['leaves'][mode]['validFromUTC']))/2).timestamp()),True) for mode in ['expired','future']]:
 cmd=['openssl','verify','-CAfile',str(R/'trusted-ca.crt'),'-purpose','sslserver','-verify_ip','192.168.129.111','-attime',str(epoch),str(R/(mode+'.crt'))];q=subprocess.run(cmd,capture_output=True,text=True,timeout=10);assert (q.returncode==0)==expected,(name,q.stdout,q.stderr);results.append({'case':name,'command':cmd,'verificationEpochUTC':datetime.datetime.fromtimestamp(epoch,datetime.timezone.utc).isoformat(),'expectedSuccess':expected,'exitCode':q.returncode,'stdout':q.stdout,'stderr':q.stderr})
(R/'offline-certificate-controls.json').write_text(json.dumps({'atUTC':now.isoformat(),'passed':True,'scope':'Offline explicit-time public-certificate checks, including valid chain/name within each leaf window','deviceClockOrTrustChanged':False,'cases':results},indent=2)+'\n');print('PASS five offline certificate date/chain/name controls',flush=True)
