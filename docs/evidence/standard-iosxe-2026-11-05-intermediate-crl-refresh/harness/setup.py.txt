import pathlib,subprocess,json,hashlib,datetime,os
r=pathlib.Path('/tmp/cvk-intrefresh-20261105');files=[]

 for p in sorted(pathlib.Path(name).iterdir()):

   b=p.read_bytes();files.append({'path':str(p),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()})
